const productos = [
  { nombre: "Cuaderno", precio: 2500, stock: 4 },
  { nombre: "Lapicera", precio: 800, stock: 0 },
  { nombre: "Mochila", precio: 18000, stock: 2 },
  { nombre: "Regla", precio: 1200, stock: 5 }
];

const disponibles = productos.filter(producto => producto.stock > 0);
const nombres = disponibles.map(producto => producto.nombre);
const productoBuscado = productos.find(producto => producto.nombre === "Mochila");

console.log("Disponibles:", nombres);
console.log("Producto buscado:", productoBuscado);
console.log("Cantidad de productos:", productos.length);

Ejercicio 1 · Conversión de datos y operadores

El código tiene errores de tipo y de comparación. ¿Cómo lo corregirías?

const precio = "1500";
const cantidad = 3;
const total = precio + cantidad;

if (total = 4503) {
  console.log("Total correcto");
} else {
  console.log("Revisar total");
}
const precio = "1500";
const cantidad = 3;
const total = Number(precio) * cantidad;

if (total === 4500) {
  console.log("Total correcto");
} else {
  console.log("Revisar total");
}

Explicación: Number() convierte el texto en número; se multiplica precio por cantidad; === compara sin conversión y evita asignar un valor dentro del if.

Resultado: Total correcto. El total es 4500.

Ejercicio 2 · Funciones y parámetros

La función debe calcular un descuento expresado en porcentaje.

function calcularDescuento(precio, porcentaje) {
  const descuento = precio * porcentaje;
  return precio - descuento;
}

const final = calcularDescuento(10000, 20);
console.log("Precio final:", final);
function calcularDescuento(precio, porcentaje) {
  const descuento = precio * (porcentaje / 100);
  return precio - descuento;
}

const final = calcularDescuento(10000, 20);
console.log("Precio final:", final);

Explicación: El porcentaje debe dividirse por 100 para convertirlo en proporción.

Resultado: Precio final: 8000.
